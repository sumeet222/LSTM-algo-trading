class FabricatedMarket:
    def __init__(self, first, last):
        self.clock = MarketClock("2020-12-01", "2022-01-01")
        self.ogn = {}
        self.first = first
        self.last = last

    def days(self):
        for date in pd.date_range(self.first, self.last):
            day = str(date.date())
            origins = self.clock.origins(day)
            if not origins:
                continue
            starts = np.arange(minute(origins[0]), minute(origins[0]) + 390)
            values = np.full((390, 20), 100.0)
            records = []
            for origin in origins:
                for i, ticker in enumerate(SYMBOLS):
                    row: dict[str, object] = dict(
                        origin=origin,
                        endpoint=self.clock.endpoint(origin),
                        ticker=ticker,
                        input_eligible=True,
                        direction=1 if i % 2 else -1,
                        checkpoint_hash="FABRICATED_MODEL",
                        issued_at=origin + pd.Timedelta(minutes=2),
                        matched_four_arms=True,
                    )
                    for arm in ARMS:
                        row["g_" + arm] = 1.0
                    records.append(row)
            yield Day(
                day,
                starts,
                values,
                values,
                np.full((390, 20), 100000.0),
                np.ones((390, 20), bool),
                pd.DataFrame(records),
                "FABRICATED_SOURCE",
                {},
            )


@pytest.mark.parametrize("lag", [1, 2, 5])
def test_wait_next_valid_and_nominal_lifetime(lag):
    o = Order("o", 0, 100 + lag, 101 + lag, (), D(-2), 100, 110)
    assert disposition(o, o.effective, o.effective + 1 + lag, lag, False) == "WAIT"
    assert disposition(o, o.effective + 1, o.effective + 2 + lag, lag, True) == "EXECUTE"
    assert disposition(o, 110, 111 + lag, lag, True) == "EXECUTE"
    assert disposition(o, 111, 112 + lag, lag, True) == "EXPIRE"
    with pytest.raises(ValueError):
        disposition(o, o.effective, o.effective + lag, lag, True)
    with pytest.raises(ValueError):
        disposition(replace(o, decision=o.effective), o.effective, 200, lag, True)
    with pytest.raises(ValueError):
        disposition(replace(o, prior_volume_available=o.decision + 1), o.effective, 200, lag, True)


def stub(monkeypatch, market):
    monkeypatch.setattr(driver, "Market", market)
    monkeypatch.setattr(driver, "load_actions", lambda: ())
    # Only a fabricated hardcoded reader remains; no historical reader survives.
    monkeypatch.setattr(release, "check_admission", lambda *args: None)


class MissingMarket(FabricatedMarket):
    def days(self):
        for day in super().days():
            valid = day.valid.copy()
            valid[4:7, 0] = False
            yield replace(day, valid=valid)


def test_real_replay_missing_minutes_advance_actual_time_and_record_hash(tmp_path, monkeypatch):
    stub(monkeypatch, MissingMarket)
    engine = driver.Driver((Scenario(),), "2021-01-04", "2021-01-05", tmp_path / "next")
    result = engine.run()
    assert not result["unresolved"]
    assert result["policy_counts"]["POSTPONED_FILL"] > 0
    with gzip.open(tmp_path / "next/baseline/events.jsonl.gz", "rt") as f:
        rows = [json.loads(s) for s in f]
    postponed = [
        r for r in rows if r["kind"] == "V5_QUALIFIED_BAR" and r["delay_scheduled_or_wall_minutes"]
    ]
    assert postponed and all(r["effective"] > r["intended_minute"] for r in postponed)
    assert all(r["source_hash"] == "FABRICATED_SOURCE" for r in postponed)
    assert all(r["V5_POLICY_VARIANT"] == POLICY for r in rows)
    for row in rows:
        expected = row.pop("hash")
        assert driver.canonical(row) == expected


def test_before_avgo_analogue_restart_is_byte_equal(tmp_path, monkeypatch):
    stub(monkeypatch, MissingMarket)
    whole = driver.Driver((Scenario(),), "2021-01-04", "2021-01-05", tmp_path / "whole")
    whole.run()
    cut = driver.Driver((Scenario(),), "2021-01-04", "2021-01-05", tmp_path / "cut")
    cut.run(stop_after=1)
    resumed = driver.Driver.restart(
        tmp_path / "cut/checkpoints/2021-01-04.json.gz", tmp_path / "cut", None
    )
    resumed.run()
    assert [a.journal_hash for a in whole.accounts] == [a.journal_hash for a in resumed.accounts]
    for kind in ("events", "daily", "cohorts"):
        with gzip.open(tmp_path / f"whole/baseline/{kind}.jsonl.gz", "rt") as f:
            a = f.read()
        with gzip.open(tmp_path / f"cut/baseline/{kind}.jsonl.gz", "rt") as f:
            b = f.read()
        assert a == b

def test_open_exit_remains_without_expiry_and_partial_fees_accumulate():
    a = Account("EQUAL", "fabricated", Scenario())
    a.lots["x"] = Lot("x", 0, 0, 240, 10, -1, -4, D(1), pending=D(-4))
    a.prices[0] = D(100)
    a.refresh()
    o = Order("partial", 0, 2, 3, (Request("x", D(-4), "ENTRY"),), D(-4), 2, 10)
    a.fill(o, D(100), 6, 3, D(".5"))
    a.fill(o, D(100), 7, 4)
    assert a.quantity[0] == -4 and a.commission == 1 and o.paid_fee == 1
    assert a.restricted() > 0 and a.cash - a.restricted() < a.initial
    exit_order = Order("exit", 0, 243, 244, (Request("x", D(4), "EXIT"),), D(4), 242, None)
    assert disposition(exit_order, 10000, 10003, 2, False) == "WAIT"
    assert disposition(exit_order, 10001, 10004, 2, True) == "EXECUTE"
    assert a.quantity[0] == -4 and a.nav() is not None
    a.reconcile()
    restored = unseal(seal(a))
    assert isinstance(restored, Account) and restored.cash == a.cash


def test_later_open_cannot_rewrite_submission_or_predecision_state():
    prior = []
    after = []
    for price in (D(100), D(10000)):
        a = Account("EQUAL", "fabricated", Scenario())
        a.prices[0] = D(100)
        a.marks_at[0] = 2
        a.lots["x"] = Lot("x", 0, 0, 240, 10, 1, 4, D(1), pending=D(4))
        a.refresh()
        o = Order("same", 0, 2, 3, (Request("x", D(4), "ENTRY"),), D(4), 2, 10)
        prior.append((seal(a), seal(o)))
        assert disposition(o, 3, 6, 2, False) == "WAIT"
        a.fill(o, price, 7, 4)
        after.append(a.cash)
        a.reconcile()
    assert prior[0] == prior[1] and after[0] != after[1]

def test_multiyear_microdollar_seconds_do_not_overflow_64bit(tmp_path, monkeypatch):
    stub(monkeypatch, FabricatedMarket)
    engine = driver.Driver((Scenario(),), "2021-01-04", "2021-01-05", tmp_path / "large")
    largest = int(np.iinfo(np.int64).max)
    engine.integrals[0] = largest - 10
    engine.last_prices[0] = 100
    engine.cursor = 1
    engine.integrate_gap(2)
    assert int(engine.integrals[0]) == largest - 10 + 100 * 1000000 * 60
    restored = unseal(seal(engine.integrals))
    assert isinstance(restored, np.ndarray)
    assert int(restored[0]) > largest
    assert (restored == engine.integrals).all()

def test_lossless_checkpoint_compression_is_atomic_deterministic_and_immutable(tmp_path):
    from lstm_data.trading.track2.stage35.final_backtest_v5.checkpoint import (
        read_checkpoint,
        write_checkpoint,
    )

    state = seal({"a": [D(".123"), 2**70, "MARKED_PENDING_NOT_UNKNOWN"]})
    one = tmp_path / "one.json.gz"
    two = tmp_path / "two.json.gz"
    write_checkpoint(one, state)
    write_checkpoint(two, state)
    assert one.read_bytes() == two.read_bytes()
    assert read_checkpoint(one) == state
    write_checkpoint(one, state)
    with pytest.raises(ValueError, match="IMMUTABLE_CHECKPOINT"):
        write_checkpoint(one, seal({"altered": True}))
